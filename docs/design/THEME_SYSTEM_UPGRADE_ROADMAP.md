## 🛠️ **TECHNICAL ARCHITECTURE**

### **🎨 Design Token System**

```typescript
// Core Design Tokens
export const designTokens = {
  colors: {
    jade: { 50: "#ECFDF5", 500: "#00A86B", 900: "#064E3B" },
    navy: { 50: "#EFF6FF", 500: "#1E3A8A", 900: "#1E3A8A" },
  },
  typography: {
    display: "Bebas Neue", // Impact headers, team names
    interface: "Inter", // UI elements, body text
    data: "IBM Plex Mono", // Statistics, player numbers
  },
  spacing: {
    square: { sm: "4px", md: "6px", lg: "8px" }, // Maximum roundness
    substantial: { base: "24px", lg: "32px" }, // Professional spacing
  },
};
```

```typescript
// Consistent theming interface
interface SquareComponentProps {
  variant?: "primary" | "secondary" | "outline" | "ghost";

  football?: boolean; // Enable football-specific styling
  technical?: boolean; // Enable technical/data styling
  professional?: boolean; // Enable coaching interface styling
}
```
